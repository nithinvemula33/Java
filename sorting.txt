---Selection Sort---

->Assume the initial element as min and swap with the actual min.
->Two loops are used.
//code.
import java.util.*;
class Main {
    public static void main(String[] args) {
        int arr[]={1,9,0,2,5,3,2,9,0,1};
        int n=arr.length;
       for(int i=0;i<n;i++)
       {
           for(int j=i+1;j<n;j++)
           {
               if(arr[i]>arr[j])
               {
                   int t=arr[i];
                   arr[i]=arr[j];
                   arr[j]=t;
               }
           }
       }System.out.println(Arrays.toString(arr));
    }
}
