---Recursion---
def : when a function calls itself until a specific condition met. 
There are two types of recursions
-->parameterised 
-->functional

**Parameterised**
-->Uses extra parameters to carry the answer.
-->Base case usually prints or returns the accumulated result.
-->Answer builts before the recursive call.
-->Easier for problems where you keep updating a result.

**Functional**
-->Returns the answer from each recursive call.
-->Base case returns a simple value.
-->Answer builts after the recursive call returns.
-->Used for problems where results are built from subproblems.

    if(i==n)
    {
        System.out.println(list);

      list.add(arr[i]);

}

//code for all possible unique subsequence .

static void subset(int i,int[] arr,List<List<Integer>> list,List<Integer> ds)
{
  int n=arr.length;
  list.add(new ArrayList<Integer>(ds));
  for(int j=i;j<n;j++)
  {
    if(j!=i&&arr[j]==arr[j-1]) continue;
    ds.add(arr[j]);
    subset(j+1,arr,list,ds);
    ds.remove(ds.size()-1);
  }
}

//code for print the subsequence where its sum =k.

static boolean subseq(ArrayList<Integer> list,int[] arr,int i,int n,int s,int sum)
{
  if(i==n)
    {
      if(sum==s)
        {
          System.out.println(list);
          return true;
        }
          else return false;
    }
      list.add(arr[i]);
      s=s+arr[i];
      if(subseq(list,arr,i+1,n,s,sum)==true) // pick
      {return true;}
        
      s=s-arr[i];
      list.remove(list.size()-1); // backtrack
        
      if(subseq(list,arr,i+1,n,s,sum)==true) //not pick
      {return true;}
      return false;
}        

//code for count the subsequence whose sum =k.

static int subseq(ArrayList<Integer> list,int[] arr,int i,int n,int s,int sum)
{
  if(i==n)
    {
      if(sum==s)
        {
          return 1;
        }
          return 0;
    }
  list.add(arr[i]);
  s=s+arr[i];
  int l=subseq(list,arr,i+1,n,s,sum);
       
  s=s-arr[i];
  list.remove(list.size()-1);
      
  int r=subseq(list,arr,i+1,n,s,sum);
        
  return l+r;
}
